-- 57) Snowflake Multi Stage Export Using Multi Task
-- CLEANUP SCRIPT

-- 1. SUSPEND TASKS
-- Stop future scheduled executions before dropping the Tasks.

ALTER TASK
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.EMPLOYEE_TASK_1
SUSPEND;

ALTER TASK
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.EMPLOYEE_TASK_2
SUSPEND;

ALTER TASK
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.EMPLOYEE_TASK_3
SUSPEND;

ALTER TASK
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.EMPLOYEE_TASK_4
SUSPEND;


-- 2. DROP TASKS
-- The Tasks are dropped first because they perform processing
-- against the Employee tables and the Internal Stages.

DROP TASK IF EXISTS
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.EMPLOYEE_TASK_1;

DROP TASK IF EXISTS
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.EMPLOYEE_TASK_2;

DROP TASK IF EXISTS
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.EMPLOYEE_TASK_3;

DROP TASK IF EXISTS
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.EMPLOYEE_TASK_4;


-- 3. DROP INTERNAL STAGES
-- Removes the four stages that store the exported CSV files.

DROP STAGE IF EXISTS
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.CUSTOMER_EXPORT_STAGE_1;

DROP STAGE IF EXISTS
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.CUSTOMER_EXPORT_STAGE_2;

DROP STAGE IF EXISTS
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.CUSTOMER_EXPORT_STAGE_3;

DROP STAGE IF EXISTS
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.CUSTOMER_EXPORT_STAGE_4;


-- 4. DROP FILE FORMATS
-- Removes the four CSV file formats used by the Internal Stages.

DROP FILE FORMAT IF EXISTS
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.CUSTOMER_EXPORT_CSV_FORMAT_1;

DROP FILE FORMAT IF EXISTS
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.CUSTOMER_EXPORT_CSV_FORMAT_2;

DROP FILE FORMAT IF EXISTS
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.CUSTOMER_EXPORT_CSV_FORMAT_3;

DROP FILE FORMAT IF EXISTS
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.CUSTOMER_EXPORT_CSV_FORMAT_4;


-- 5. DROP EMPLOYEE TABLES
-- Removes the four source tables.

DROP TABLE IF EXISTS
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.EMPLOYEE_DATA_1;

DROP TABLE IF EXISTS
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.EMPLOYEE_DATA_2;

DROP TABLE IF EXISTS
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.EMPLOYEE_DATA_3;

DROP TABLE IF EXISTS
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA.EMPLOYEE_DATA_4;


-- 6. DROP SCHEMA

DROP SCHEMA IF EXISTS
    SNOWFLAKE_TASK_PRACTICE.TASK_SCHEMA;


-- 7. DROP DATABASE

DROP DATABASE IF EXISTS
    SNOWFLAKE_TASK_PRACTICE;


-- 8. DROP WAREHOUSE

DROP WAREHOUSE IF EXISTS TASK_WH;
